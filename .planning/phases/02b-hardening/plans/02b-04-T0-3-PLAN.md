# scripts/lib/transition.py
"""ADR-001 Transition Table (G2-B). Single source of truth for phase set validation."""
from __future__ import annotations

class InvalidTransition(Exception):
    """Raised when a (current, target) pair violates ADR-001."""

VALID_PHASES = ("discuss", "plan", "execute", "done")

# Cell value: ("ok" | "needs_approval" | "needs_reset" | "restamp" | "invalid")
TRANSITION_TABLE: dict[tuple[str | None, str], str] = {
    (None, "discuss"): "ok",
    ("discuss", "discuss"): "ok",
    ("discuss", "plan"): "ok",
    ("plan", "discuss"): "needs_reset",
    ("plan", "plan"): "ok",
    ("plan", "execute"): "needs_approval",
    ("execute", "discuss"): "needs_reset",
    ("execute", "plan"): "needs_reset",
    ("execute", "execute"): "ok",
    ("execute", "done"): "needs_approval",
    ("done", "discuss"): "needs_reset",
    ("done", "plan"): "needs_reset",
    ("done", "execute"): "needs_reset",
    ("done", "done"): "restamp",
}

def validate_transition(*, current: str | None, target: str, approved: bool, reset_approval: bool) -> None:
    if target not in VALID_PHASES:
        raise InvalidTransition(
            f"error: cannot set phase={target} from phase={current} (see ADR-001 transition table). "
            f"Transition is undefined; choose discuss/plan/execute/done as the next step."
        )
    rule = TRANSITION_TABLE.get((current, target), "invalid")
    if rule == "ok" or rule == "restamp":
        return
    if rule == "needs_approval":
        if approved:
            return
        raise InvalidTransition(
            f"error: cannot set phase={target} from phase={current} (see ADR-001 transition table). "
            f"Run 'harness phase approve' first."
        )
    if rule == "needs_reset":
        if reset_approval:
            return
        raise InvalidTransition(
            f"error: cannot set phase={target} from phase={current} (see ADR-001 transition table). "
            f"Pass --reset-approval to clear prior approval and proceed."
        )
    raise InvalidTransition(
        f"error: cannot set phase={target} from phase={current} (see ADR-001 transition table). "
        f"Transition is undefined; choose discuss/plan/execute/done as the next step."
    )
```

# scripts/lib/session.py
"""Session lockfile per ADR-003a G1-B. Posix-only (fcntl); Windows out of scope (spec non-goal)."""
from __future__ import annotations

import atexit
import contextlib
import fcntl
import json
import os
import platform
import signal
import socket
import sys
import time
from pathlib import Path

from lib.exitcodes import EXIT_LOCKED

class LockfileExists(Exception):
    pass

_active_lockfile: Path | None = None

def read_boot_id() -> str | None:
    if platform.system() != "Linux":
        return None
    try:
        return Path("/proc/sys/kernel/random/boot_id").read_text().strip()
    except OSError:
        return None

def _now_iso_nanos() -> str:
    ns = time.time_ns()
    secs, rem = divmod(ns, 1_000_000_000)
    import datetime
    base = datetime.datetime.fromtimestamp(secs, tz=datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%S")
    return f"{base}.{rem:09d}Z"

def _harness_version() -> str:
    try:
        import harness
        return getattr(harness, "HARNESS_VERSION", "0.0.0-dev+unknown")
    except Exception:
        return "0.0.0-dev+unknown"

def _build_payload() -> dict:
    return {
        "pid": os.getpid(),
        "hostname": socket.gethostname(),
        "started_at_utc": _now_iso_nanos(),
        "harness_version": _harness_version(),
        "boot_id": read_boot_id(),
    }

def release_lock(lock_path: Path | None = None) -> None:
    global _active_lockfile
    target = lock_path or _active_lockfile
    if target is None:
        return
    try:
        target.unlink()
    except FileNotFoundError:
        pass
    if target == _active_lockfile:
        _active_lockfile = None

def _install_signal_handlers(lock_path: Path) -> None:
    def _handler(signum, frame):
        release_lock(lock_path)
        # Re-raise default behavior
        signal.signal(signum, signal.SIG_DFL)
        os.kill(os.getpid(), signum)
    signal.signal(signal.SIGINT, _handler)
    signal.signal(signal.SIGTERM, _handler)

@contextlib.contextmanager
def acquire_lock(*, lock_path: Path):
    """Acquire the session lock via O_EXCL + flock. Raises LockfileExists on contention."""
    global _active_lockfile
    lock_path.parent.mkdir(parents=True, exist_ok=True)
    try:
        fd = os.open(str(lock_path), os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o644)
    except FileExistsError as exc:
        raise LockfileExists(str(lock_path)) from exc
    try:
        try:
            fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
        except OSError:
            os.close(fd)
            try: lock_path.unlink()
            except FileNotFoundError: pass
            raise LockfileExists(str(lock_path))
        payload_bytes = (json.dumps(_build_payload()) + "\n").encode("utf-8")
        os.write(fd, payload_bytes)
        os.fsync(fd)
    finally:
        try: os.close(fd)
        except OSError: pass
    _active_lockfile = lock_path
    atexit.register(release_lock, lock_path)
    _install_signal_handlers(lock_path)
    try:
        yield lock_path
    finally:
        release_lock(lock_path)

def read_lock_payload(lock_path: Path) -> dict:
    return json.loads(lock_path.read_text())

def is_pid_alive(pid: int) -> bool:
    try:
        os.kill(pid, 0)
        return True
    except ProcessLookupError:
        return False
    except PermissionError:
        return True  # Process exists but we can't signal it
```

# scripts/lib/audit.py
"""Audit log per ADR-003a G1-A. Atomic append via flock, PIPE_BUF-safe lines, rotation at 10 MiB or 10000 entries."""
from __future__ import annotations

import fcntl
import hashlib
import json
import os
from pathlib import Path

AUDIT_MAX_LINE_BYTES = 512  # macOS PIPE_BUF
ROTATION_BYTES = 10 * 1024 * 1024
ROTATION_ENTRIES = 10000
ROTATION_KEEP = 5

def compute_state_hash(state_path: Path) -> str:
    if not state_path.exists():
        return ""
    return hashlib.sha256(state_path.read_bytes()).hexdigest()

def _read_last_index(audit_path: Path) -> int:
    if not audit_path.exists() or audit_path.stat().st_size == 0:
        return 0
    with audit_path.open("rb") as f:
        f.seek(0, os.SEEK_END)
        size = f.tell()
        # Scan backward to find last newline
        chunk = 1024
        pos = size
        buf = b""
        while pos > 0:
            read = min(chunk, pos)
            pos -= read
            f.seek(pos)
            buf = f.read(read) + buf
            if b"\n" in buf.rstrip(b"\n"):
                break
        lines = buf.split(b"\n")
        for ln in reversed(lines):
            if ln.strip():
                try:
                    return int(json.loads(ln)["index"])
                except (ValueError, KeyError):
                    continue
    return 0

def read_last_entry(audit_path: Path) -> dict | None:
    if not audit_path.exists() or audit_path.stat().st_size == 0:
        return None
    text = audit_path.read_text()
    for ln in reversed(text.splitlines()):
        if ln.strip():
            return json.loads(ln)
    return None

def _rotate(audit_path: Path) -> None:
    # Caller holds the flock; rename audit.log.N -> audit.log.N+1, then audit.log -> audit.log.1
    for n in range(ROTATION_KEEP, 0, -1):
        src = audit_path.with_name(f"{audit_path.name}.{n}")
        if n == ROTATION_KEEP and src.exists():
            src.unlink()
            continue
        dst = audit_path.with_name(f"{audit_path.name}.{n+1}")
        if src.exists():
            os.rename(src, dst)
    os.rename(audit_path, audit_path.with_name(f"{audit_path.name}.1"))

def _write_overflow(audit_path: Path, index: int, full_entry: dict) -> None:
    overflow_dir = audit_path.parent / "audit.overflow"
    overflow_dir.mkdir(parents=True, exist_ok=True)
    (overflow_dir / f"{index}.json").write_text(json.dumps(full_entry, indent=2) + "\n")

def audit_append(entry: dict, *, audit_path: Path) -> int:
    """Append a single JSON line to audit_path with flock + PIPE_BUF safety. Returns the index written."""
    audit_path.parent.mkdir(parents=True, exist_ok=True)
    fd = os.open(str(audit_path), os.O_WRONLY | os.O_APPEND | os.O_CREAT, 0o644)
    try:
        fcntl.flock(fd, fcntl.LOCK_EX)
        # Rotation check (size + entry count)
        st_size = os.fstat(fd).st_size
        last_idx = _read_last_index(audit_path)
        if st_size >= ROTATION_BYTES or last_idx >= ROTATION_ENTRIES:
            fcntl.flock(fd, fcntl.LOCK_UN)
            os.close(fd)
            _rotate(audit_path)
            fd = os.open(str(audit_path), os.O_WRONLY | os.O_APPEND | os.O_CREAT, 0o644)
            fcntl.flock(fd, fcntl.LOCK_EX)
            last_idx = 0
        index = last_idx + 1
        full_entry = dict(entry, index=index)
        line = json.dumps(full_entry, separators=(",", ":")) + "\n"
        if len(line.encode("utf-8")) > AUDIT_MAX_LINE_BYTES:
            _write_overflow(audit_path, index, full_entry)
            truncated = dict(entry, index=index, args={"truncated": True})
            line = json.dumps(truncated, separators=(",", ":")) + "\n"
        assert len(line.encode("utf-8")) <= AUDIT_MAX_LINE_BYTES, "overflow even after truncation"
        os.write(fd, line.encode("utf-8"))
        os.fsync(fd)
    finally:
        try:
            fcntl.flock(fd, fcntl.LOCK_UN)
            os.close(fd)
        except OSError:
            pass
    return index
```

def _now_iso_nanos() -> str:
    import time, datetime
    ns = time.time_ns()
    secs, rem = divmod(ns, 1_000_000_000)
    base = datetime.datetime.fromtimestamp(secs, tz=datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%S")

def _identify() -> str:
    import subprocess
    try:
        r = subprocess.run(["git","config","user.email"], capture_output=True, text=True, timeout=2)
        if r.returncode == 0 and r.stdout.strip():
            return r.stdout.strip()
    except Exception:
        pass
    return os.environ.get("USER", "unknown")

def _load_state() -> tuple[dict, str | None]:
    if not STATE_PATH.exists():
        return ({}, None)

        data = json.loads(STATE_PATH.read_text())
    except json.JSONDecodeError as exc:
        print(f"error: .scratch/phase-state.json is unparseable ({exc}); fix the JSON or restore from a backup before retrying.", file=sys.stderr)
        sys.exit(EXIT_UNPARSEABLE_JSON)

    if not raw.strip():
        return {}
    try:
        return json.loads(raw)
    except json.JSONDecodeError as exc:
        print(f"error: --stdin-json input is unparseable ({exc}); pass valid JSON via stdin.", file=sys.stderr)
        sys.exit(EXIT_UNPARSEABLE_JSON)

    state, current = _load_state()
    before_hash = compute_state_hash(STATE_PATH)
    target = args.phase
    approved = bool(state.get("approved"))
    try:
        validate_transition(current=current, target=target, approved=approved,
                            reset_approval=args.reset_approval)
    except InvalidTransition as exc:
        print(str(exc), file=sys.stderr)
        return EXIT_INVALID_TRANSITION

    new_state = dict(state)
    new_state["phase"] = target
    new_state["updated_at"] = _now_iso_nanos()
    new_state["updated_by"] = _identify()
    if args.reset_approval:
        new_state["approved"] = False
        new_state["approved_by"] = None
        new_state["approved_at"] = None
    if args.plan_id is not None:
        new_state["plan_id"] = args.plan_id
    if args.summary is not None:
        new_state["summary"] = args.summary
    if args.stdin_json:
        extra = _read_stdin_json()
        for k, v in extra.items():
            if k in ALLOWED_STDIN_FIELDS:
                new_state[k] = v

    _write_state_atomic(new_state)
    after_hash = compute_state_hash(STATE_PATH)
    cli_args = {"phase": target}
    if args.plan_id: cli_args["plan_id"] = args.plan_id
    if args.summary: cli_args["summary"] = args.summary
    if args.reset_approval: cli_args["reset_approval"] = True
    verb_name = "phase.set.noop" if current == target else "phase.set"
    idx = audit_append({
        "verb": verb_name, "args": cli_args,
        "before_sha256": before_hash, "after_sha256": after_hash,

    }, audit_path=AUDIT_PATH)
    print(json.dumps({
        "ok": True, "verb": "phase.set", "previous_phase": current,
        "phase": target, "state_path": str(STATE_PATH),
        "audit_entry_index": idx, "updated_at": new_state["updated_at"],
        "updated_by": new_state["updated_by"],
    }, indent=2))
    return EXIT_OK

import datetime

def _parse_at(at_str: str) -> str:
    try:
        # Strip trailing Z for fromisoformat
        normalized = at_str.replace("Z","+00:00")
        dt = datetime.datetime.fromisoformat(normalized)
    except ValueError as exc:
        print(f"error: --at value {at_str!r} could not be parsed as ISO-8601 ({exc}).", file=sys.stderr)
        sys.exit(EXIT_TIMESTAMP_RANGE)
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=datetime.timezone.utc)
    now = datetime.datetime.now(datetime.timezone.utc)
    if abs((now - dt).total_seconds()) > 86400:
        print(f"error: --at value {at_str!r} is not within 24h of current UTC time ({now.isoformat()}); refusing to write a far-future or far-past timestamp.", file=sys.stderr)
        sys.exit(EXIT_TIMESTAMP_RANGE)

    current_boot = read_boot_id()
    recorded_boot = payload.get("boot_id")
    if recorded_boot is not None and current_boot is not None and recorded_boot != current_boot:
        # Boot ID changed → host rebooted, pid is definitely stale.
        LOCK_PATH.unlink()
        return EXIT_OK
    if recorded_boot is None and current_boot is None:
        # macOS: cannot verify boot_id; if pid is dead we already returned ok above
        LOCK_PATH.unlink()
        return EXIT_OK
    LOCK_PATH.unlink()
    return EXIT_OK
```
